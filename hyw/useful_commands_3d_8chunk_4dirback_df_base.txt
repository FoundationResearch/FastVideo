export ACTION_CKPT_PIPE=/mnt/weka/home/hao.zhang/alex/weights/tencent/HY-WorldPlay/ar_model/diffusion_pytorch_model.safetensors

# -----------------------------------------------------------------------------
# 1. 生成 sythball 3D 数据 (4dirback, 4 samples, 125 frames each; == 8 AR chunks)
# -----------------------------------------------------------------------------

python -m hyw.sythgenerator sythball \
  --out_root /mnt/weka/home/hao.zhang/alex/FastVideo/hyw/data/sythball_v1_125f_8chunk_4dirback \
  --split train --num_samples 4 --num_frames 125 --fps 25 --width 256 --height 256 --seed 0 \
  --hold_action_frames 4 --fixed_move_action_mode 4dirback --fixed_view_action ""

# （可选）test set：用于 eval
python -m hyw.sythgenerator sythball \
  --out_root /mnt/weka/home/hao.zhang/alex/FastVideo/hyw/data/sythball_v1_125f_noview \

  --macro_period 4 \
  --hold_action_frames 4 --fixed_view_action ""

# 4dirdouble test
python -m hyw.sythgenerator sythball \

# test (可选)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_125f_noview/manifest_raw_test.json \
  --model_path "${MODEL_PATH}" \
  --action_ckpt "${ACTION_CKPT_PIPE}" \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_125f_noview_modelinput/latent_pt/test \
