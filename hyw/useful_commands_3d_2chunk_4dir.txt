
# =============================================================================
# sythball 3D数据 (2chunk/29f) - 4dir 数据集完整训练流程
#   与 simple 唯一差别：train set 生成 4 个样本（前后左右），对应 move_action = W/S/A/D
# =============================================================================
#
# 关键对齐（官方实现）：
# - VAE 时间压缩比 temporal_compression_ratio=4
# - latent 帧数 L = (F-1)//4 + 1
# - “2 个 chunk”最稳：L=8（=2*4），对应 F = 4*(8-1)+1 = 29
# - 推理/训练按 latent 维度切 chunk：chunk_latent_frames=4，因此 chunk_num=2


# -----------------------------------------------------------------------------
# 0. 环境变量 (根据你的机器调整)
# -----------------------------------------------------------------------------
export MODEL_PATH=/mnt/weka/home/hao.zhang/alex/weights/tencent/HunyuanVideo-1.5
export ACTION_CKPT=/mnt/weka/home/hao.zhang/alex/weights/tencent/HY-WorldPlay/ar_model/diffusion_pytorch_model.safetensors


# -----------------------------------------------------------------------------
# 1. 生成 sythball 3D 数据 (4dir, 4 samples, 29 frames each; == 2 AR chunks)
# -----------------------------------------------------------------------------
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_29f_2chunk_4dir \
  --split train --num_samples 4 --num_frames 29 --fps 25 --width 256 --height 256 --seed 0 \
  --hold_action_frames 4 --fixed_move_action_mode 4dir --fixed_view_action ""

# （可选）test set：用于 eval（和 1chunk_4dir 一样，用随机动作 + 禁止 view rotation 即可）
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_29f_noview \
  --split test --num_samples 4 --num_frames 29 --fps 25 --width 256 --height 256 --seed 1000 \
  --macro_period 4 \
  --hold_action_frames 4 --fixed_view_action ""


# -----------------------------------------------------------------------------
# 2. 预计算 latents (VAE encode + prompt embeds)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# train (4dir)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_29f_2chunk_4dir/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_29f_2chunk_4dir_modelinput/latent_pt/train \
  --max_samples 4

# test (可选，用于 eval)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_29f_noview/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_29f_noview_modelinput/latent_pt/test \
  --max_samples 4


# -----------------------------------------------------------------------------
# 3. 生成训练 JSON (组合 latent_path + pose_path + action_path)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

python train/make_training_json.py \
  --raw_manifest data/sythball_v1_29f_2chunk_4dir/manifest_raw_train.json \
  --latent_root data/sythball_v1_29f_2chunk_4dir_modelinput/latent_pt/train \
  --out_json data/sythball_v1_29f_2chunk_4dir_modelinput/sythball_v1_29f_2chunk_4dir_train_for_hyworld.json \
  --max_samples 4


# -----------------------------------------------------------------------------
# 4. 训练（2chunk/29f）
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw
#
# 2 chunks 对应：
#   NUM_FRAMES=29
#   NUM_LATENT_T=8
#   WINDOW_FRAMES=8   (window_frames 是 latent 维度长度；这里覆盖 2 个 chunk)
#
# NOTE:
# - 本 4dir train set 只有 4 条样本；当 NUM_GPUS=4 且 --train-batch-size=1 时，
#   每个 step 实际会并行消耗 4 条样本（每张卡 1 条），因此会看到 “1 step = 1 epoch”。
# - 这是正常的；想看到每个 epoch 多个 step，可以把 NUM_GPUS 改小（如 1）或增大 num_samples。
TRAIN_JSON=~/alex/FastVideo/hyw/data/sythball_v1_29f_2chunk_4dir_modelinput/sythball_v1_29f_2chunk_4dir_train_for_hyworld.json \
OUT_DIR=~/alex/FastVideo/hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small \
WANDB_RUN_NAME=hyworld_sythball_29f_2chunk_4dir_small \
NUM_FRAMES=29 NUM_LATENT_T=8 WINDOW_FRAMES=8 \
NUM_GPUS=4 CUDA_VISIBLE_DEVICES=4,1,2,3 \
MAX_TRAIN_STEPS=2000 CHECKPOINTING_STEPS=500 \
TRAIN_VIDEO_LOG_STEPS=25 TRAIN_VIDEO_LOG_MAX_SAMPLES=1 TRAIN_VIDEO_LOG_FPS=25 \
MASTER_PORT=29612 \
bash train/run_train_small.sh


# -----------------------------------------------------------------------------
# 5. 评估 (使用 eval/eval_sythcircle_test.py，它是通用的)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo

# Base model eval（用于对比“训练前 vs 训练后”）
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_29f_2chunk_4dir/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --out_dir hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small/eval_base \
  --num_inference_steps 50 \
  --guidance_scale 1.0 \
  --flow_shift 3.0 \
  --seed 42 \
  --model_type ar \
  --max_samples 4

# Finetuned ckpt eval（把 checkpoint-XXX 改成你想测的）
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_29f_2chunk_4dir/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --finetuned_ckpt hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small/checkpoint-2000 \
  --out_dir hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small/eval_ckpt2000-3.0shift-ar \
  --num_inference_steps 50 \
  --guidance_scale 1.0 \
  --flow_shift 3.0 \
  --seed 42 \
  --model_type ar \
  --max_samples 4

# （可选）no-view 测试集 eval（检验泛化；先跑第 1 步生成 test set）
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_29f_noview/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --finetuned_ckpt hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small/checkpoint-2000 \
  --out_dir hyw/outputs/hyworld_sythball_29f_2chunk_4dir_small/eval_ckpt2000-3.0shift-ar-noview-test \
  --num_inference_steps 50 \
  --guidance_scale 1.0 \
  --flow_shift 3.0 \
