export MODEL_PATH=/mnt/weka/home/hao.zhang/alex/weights/tencent/HunyuanVideo-1.5

# -----------------------------------------------------------------------------
# 1. 生成 sythball 3D 数据
# -----------------------------------------------------------------------------
# train set (32 samples, 13 frames each; == 1 AR chunk in latent space)
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_13f_simple \
  --split train --num_samples 1 --num_frames 13 --fps 25 --width 256 --height 256 --seed 0 \
  --hold_action_frames 4 --fixed_move_action W --fixed_view_action ""
#
# Simple/debug mode (fixed action, no view rotation):
#   --fixed_move_action W --fixed_view_action ""

# test set (16 samples)
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_13f \
  --split test --num_samples 16 --num_frames 13 --fps 25 --width 256 --height 256 --seed 1000 \
  --hold_action_frames 4

# -----------------------------------------------------------------------------
# 2. 预计算 latents (VAE encode + prompt embeds)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# train
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_13f_simple/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_13f_simple_modelinput/latent_pt/train \
  --max_samples 32

# test (可选，用于 eval)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_13f/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_13f_modelinput/latent_pt/test \
  --max_samples 16

# -----------------------------------------------------------------------------
# 3. 生成训练 JSON (组合 latent_path + pose_path + action_path)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# train
python train/make_training_json.py \
  --raw_manifest data/sythball_v1_13f_simple/manifest_raw_train.json \
  --latent_root data/sythball_v1_13f_simple_modelinput/latent_pt/train \
  --out_json data/sythball_v1_13f_simple_modelinput/sythball_v1_13f_train_for_hyworld.json \
  --max_samples 32

# test (可选)
python train/make_training_json.py \
  --raw_manifest data/sythball_v1_13f/manifest_raw_test.json \
  --latent_root data/sythball_v1_13f_modelinput/latent_pt/test \
  --out_json data/sythball_v1_13f_modelinput/sythball_v1_13f_test_for_hyworld.json \
  --max_samples 16

# -----------------------------------------------------------------------------
# 4. 训练
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# 方式A: 使用环境变量覆盖默认配置

# -----------------------------------------------------------------------------
# 6. 评估 (使用 eval/eval_sythcircle_test.py，它是通用的)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo

# Base model（不加载 finetuned_ckpt）做 eval：用于对比“训练前 vs 训练后”
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_13f_simple/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --out_dir hyw/outputs/hyworld_sythball_13f_simple_small/eval_base \
  --num_inference_steps 50 \
  --guidance_scale 1.0 \
  --flow_shift 3.0 \
  --model_type bi \
  --max_samples 1 \
  --seed 42

# 用训练数据做 eval (overfit 测试) - 推荐先用这个检查 loss 是否真正下降
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_13f_simple/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --finetuned_ckpt hyw/outputs/hyworld_sythball_13f_simple_small/checkpoint-500 \
  --out_dir hyw/outputs/hyworld_sythball_13f_simple_small/eval_ckpt500-3.0shift-ar \
  --num_inference_steps 50 \
  --guidance_scale 1.0 \
  --max_samples 1 \
  --flow_shift 3.0 \
  --model_type ar \
  --print_weight_delta \
  --seed 42

# -----------------------------------------------------------------------------
# 6b. 诊断：用 training(transformer) 的 forward 跑完整 50-step denoising（对齐训练逻辑）
# -----------------------------------------------------------------------------
# 说明：
# - 这条命令使用 HY-WorldPlay trainer 侧 transformer（训练时用的那套）做 50 step 推理
# - 输入 packing 使用 (x_t || cond_latents || cond_mask) => 2*C+1 通道（C=32 -> 65）
# - 自动初始化单进程的 SP/TP/DP group（避免 `_SP is None`）
python -m hyw.eval.infer_with_trainer_transformer_50step \
  --train_json hyw/data/sythball_v1_13f_simple_modelinput/sythball_v1_13f_train_for_hyworld.json \
  --sample_idx 0 \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --finetuned_ckpt hyw/outputs/hyworld_sythball_13f_simple_small/checkpoint-500 \
  --out_dir hyw/outputs/hyworld_sythball_13f_simple_small/infer_trainer_50step_ckpt500 \
  --num_inference_steps 50 \
  --flow_shift 3.0 \
  --dtype bf16 \
  --seed 42 \
  --fps 25

# 用测试数据做 eval (泛化能力测试) - 需要先生成 test 数据
python hyw/eval/eval_sythcircle_test.py \
  --raw_manifest hyw/data/sythball_v1_13f/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --finetuned_ckpt hyw/outputs/hyworld_sythball_13f_small/checkpoint-100 \

  --max_samples 4 \
