
# =============================================================================
# sythball 3D数据 (8chunk/125f) - 4dir 数据集完整训练流程
#   目标：让训练真正进入 out-window + memory selection（FOV overlap）分支
# =============================================================================
#
# 关键对齐（官方实现）：
# - VAE 时间压缩比 r=4（对应 latent_T = (F-1)//4 + 1）
# - 1 chunk = 4 个 latent（AR 的 chunk-wise causal / memory 选择均以 4 对齐）
# - 8 chunks => latent_T = 8*4 = 32
# - 对应原视频帧数：F = 4*(32-1)+1 = 125
#
# 为什么选 125 帧：
# - latent_T=32 足够长：dataset 的 out-window 能采到 current_frame_idx ∈ {16,20,24,28}
# - 并且 current_frame_idx=24/28 > memory_frames(20)，会触发基于 FOV overlap 的“真正 memory 挑选”
#
# 建议训练参数：
# - WINDOW_FRAMES=16（=4 chunks）：既能进入 out-window，又能让 memory 选择生效
#


# -----------------------------------------------------------------------------
# 0. 环境变量 (根据你的机器调整)
# -----------------------------------------------------------------------------
export MODEL_PATH=/mnt/weka/home/hao.zhang/alex/weights/tencent/HunyuanVideo-1.5
export ACTION_CKPT=/mnt/weka/home/hao.zhang/alex/weights/tencent/HY-WorldPlay/ar_model/diffusion_pytorch_model.safetensors


# -----------------------------------------------------------------------------
# 1. 生成 sythball 3D 数据 (4dir, 4 samples, 125 frames each; == 8 AR chunks)
# -----------------------------------------------------------------------------
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_125f_8chunk_4dir \
  --split train --num_samples 4 --num_frames 125 --fps 25 --width 256 --height 256 --seed 0 \
  --hold_action_frames 4 --fixed_move_action_mode 4dir --fixed_view_action ""

# （可选）test set：用于 eval（禁用 view rotation，随机动作即可）
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_125f_noview \
  --split test --num_samples 8 --num_frames 125 --fps 25 --width 256 --height 256 --seed 1000 \
  --macro_period 4 \
  --hold_action_frames 4 --fixed_view_action ""


# -----------------------------------------------------------------------------
# 2. 预计算 latents (VAE encode + prompt embeds)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

  --raw_manifest data/sythball_v1_125f_8chunk_4dir/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \

  --out_root data/sythball_v1_125f_8chunk_4dir_modelinput/latent_pt/train \
  --max_samples 4

# test (可选)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_125f_noview/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_125f_noview_modelinput/latent_pt/test \
  --max_samples 8


# -----------------------------------------------------------------------------
# 3. 生成训练 JSON (组合 latent_path + pose_path + action_path)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

python train/make_training_json.py \
  --raw_manifest data/sythball_v1_125f_8chunk_4dir/manifest_raw_train.json \
  --latent_root data/sythball_v1_125f_8chunk_4dir_modelinput/latent_pt/train \
  --out_json data/sythball_v1_125f_8chunk_4dir_modelinput/sythball_v1_125f_8chunk_4dir_train_for_hyworld.json \
  --max_samples 4


# -----------------------------------------------------------------------------
# 4. 训练（8chunk/125f）
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw
#
# 8 chunks 对应：
#   NUM_FRAMES=125
#   NUM_LATENT_T=32
#   WINDOW_FRAMES=16   (latent 维度；推荐 16 来体验 in-window/out-window + memory selection)
#
# NOTE:
# - 4dir train set 只有 4 条样本；NUM_GPUS=4 且 batch=1 时可能出现 “1 step = 1 epoch”，正常现象。
# - 如果你想更稳定地观察 out-window/memory 分支日志，可以把 num_samples 加大（比如 32/64）。
TRAIN_JSON=~/alex/FastVideo/hyw/data/sythball_v1_125f_8chunk_4dir_modelinput/sythball_v1_125f_8chunk_4dir_train_for_hyworld.json \
OUT_DIR=~/alex/FastVideo/hyw/outputs/hyworld_sythball_125f_8chunk_4dir_small \
WANDB_RUN_NAME=hyworld_sythball_125f_8chunk_4dir_small \

NUM_GPUS=4 CUDA_VISIBLE_DEVICES=0,1,2,3 \
MAX_TRAIN_STEPS=2000 CHECKPOINTING_STEPS=500 \
TRAIN_VIDEO_LOG_STEPS=25 TRAIN_VIDEO_LOG_MAX_SAMPLES=1 TRAIN_VIDEO_LOG_FPS=25 \
