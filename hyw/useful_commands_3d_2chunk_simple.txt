
# =============================================================================
# sythball 3D数据：2-chunk（AR）训练流程（simple/debug）
# =============================================================================
#
# 关键对齐（官方实现）：
# - VAE 时间压缩比 temporal_compression_ratio=4
# - latent 帧数 L = (F-1)//4 + 1
# - HY-WorldPlay 的 AR rollout 按 latent 维度切 chunk：chunk_latent_frames=4
# - 推理侧：chunk_num = latent_frames // chunk_latent_frames
# - 训练侧：dataset 会把 latent 长度截到 4 的倍数：max_length = latent_length//4*4
#
# 因此，“2 个 chunk”最稳的设置是：
#   L = 8  (== 2 * 4)
#   F = 4*(L-1)+1 = 29
#
# 注意：如果你用 F=30/31/32，依然会 encode 成 L=8，但最后几帧会被 VAE/pose 对齐逻辑丢弃。


# -----------------------------------------------------------------------------
# 0. 环境变量 (根据你的机器调整)
# -----------------------------------------------------------------------------
export MODEL_PATH=/mnt/weka/home/hao.zhang/alex/weights/tencent/HunyuanVideo-1.5
export ACTION_CKPT=/mnt/weka/home/hao.zhang/alex/weights/tencent/HY-WorldPlay/ar_model/diffusion_pytorch_model.safetensors


# -----------------------------------------------------------------------------
# 1. 生成 sythball 3D 数据（2 chunks）
# -----------------------------------------------------------------------------
# train set (debug: 1 sample; 29 frames -> latent_T=8 -> 2 chunks)
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_29f_2chunk_simple \
  --split train --num_samples 1 --num_frames 29 --fps 25 --width 256 --height 256 --seed 0 \
  --hold_action_frames 4 --fixed_move_action W --fixed_view_action ""

# test set (可选，用于 eval；建议多一些样本看泛化)
python -m hyw.sythgenerator sythball \
  --out_root ~/alex/FastVideo/hyw/data/sythball_v1_29f_2chunk \
  --split test --num_samples 16 --num_frames 29 --fps 25 --width 256 --height 256 --seed 1000 \
  --hold_action_frames 4


# -----------------------------------------------------------------------------
# 2. 预计算 latents (VAE encode + prompt embeds)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# train
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_29f_2chunk_simple/manifest_raw_train.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_29f_2chunk_simple_modelinput/latent_pt/train \
  --max_samples 32

# test (可选)
python train/precompute_latents.py \
  --raw_manifest data/sythball_v1_29f_2chunk/manifest_raw_test.json \
  --model_path $MODEL_PATH \
  --action_ckpt $ACTION_CKPT \
  --transformer_version 480p_i2v \
  --out_root data/sythball_v1_29f_2chunk_modelinput/latent_pt/test \
  --max_samples 16


# -----------------------------------------------------------------------------
# 3. 生成训练 JSON (组合 latent_path + pose_path + action_path)
# -----------------------------------------------------------------------------
cd ~/alex/FastVideo/hyw

# train
python train/make_training_json.py \
  --raw_manifest data/sythball_v1_29f_2chunk_simple/manifest_raw_train.json \
  --latent_root data/sythball_v1_29f_2chunk_simple_modelinput/latent_pt/train \
  --out_json data/sythball_v1_29f_2chunk_simple_modelinput/sythball_v1_29f_2chunk_train_for_hyworld.json \
  --max_samples 32

# test (可选)
python train/make_training_json.py \
  --raw_manifest data/sythball_v1_29f_2chunk/manifest_raw_test.json \
  --latent_root data/sythball_v1_29f_2chunk_modelinput/latent_pt/test \
  --out_json data/sythball_v1_29f_2chunk_modelinput/sythball_v1_29f_2chunk_test_for_hyworld.json \
  --max_samples 16


# -----------------------------------------------------------------------------
# 4. 训练（AR, 2 chunks）
# -----------------------------------------------------------------------------
# 2 chunks 对应：
#   NUM_LATENT_T=8
#   NUM_FRAMES=29
#   WINDOW_FRAMES=8   (window_frames 是 latent 维度长度；这里就是“窗口覆盖 2 chunks”)
cd ~/alex/FastVideo/hyw

TRAIN_JSON=~/alex/FastVideo/hyw/data/sythball_v1_29f_2chunk_simple_modelinput/sythball_v1_29f_2chunk_train_for_hyworld.json \
OUT_DIR=~/alex/FastVideo/hyw/outputs/hyworld_sythball_29f_2chunk_simple_small \
WANDB_RUN_NAME=hyworld_sythball_29f_2chunk_simple_small \
NUM_FRAMES=29 NUM_LATENT_T=8 WINDOW_FRAMES=8 \
NUM_GPUS=1 CUDA_VISIBLE_DEVICES=0 \
MAX_TRAIN_STEPS=2000 CHECKPOINTING_STEPS=500 \
TRAIN_VIDEO_LOG_STEPS=25 TRAIN_VIDEO_LOG_MAX_SAMPLES=1 TRAIN_VIDEO_LOG_FPS=25 \
bash train/run_train_small.sh

# NOTE:
# - 数据很小（1 sample）时用 1 GPU 更稳（避免 drop_last 导致 0 batch）。
# - 2 chunks 的短序列不支持 “outside-window chunk” 采样（latent_T 太短），这是预期行为。
